# Deliberately extreme speed, to check the game is reading mods.
# Includes the momentum fixes from faster-swinging.txt. Gravity is normal.
# Not meant for normal play.

[configs\hero\hero_swingconfig.config]
# Top swing speed and build-up. Applied to ALL swing contexts now (standard,
# fast, near-ground, indoor) so speed no longer drops off low to the ground or
# past buildings. That drop-off was the "sometimes it gets slow".
MotionParams*.SpeedParams.TerminalVelocity* *= 3
MotionParams*.SpeedParams.SpeedBoostAccel *= 3
MotionParams*.SpeedParams.SpeedBoostTarget *= 3
MomentumConfig.*BoostCap *= 3

# Speed coming out of a swing (letting go, or jumping off the web).
MotionParams*.SpeedParams.ReleaseBoostXZ *= 2.5
MotionParams*.SpeedParams.JumpBoostXZ *= 2.5
MomentumConfig.SwingReleaseData.ReleaseData.Boost *= 3
MomentumConfig.SwingReleaseData.JumpData.Boost *= 2

# Keep all incoming speed when a swing starts (standard/fast sets have this).
MotionParams*.SpeedParams.InSpeedCarryover = 1
JumpFromFallHorzDamp = 2
# Air momentum decay rate (0.4 was 40% of normal; 0.8 fades a bit faster).
MomentumConfig.DecayDataAir[*].DecayRate *= 0.8
MomentumConfig.DecayDataAir[*].DecayDelay *= 2

# Experiment: slightly stronger fall gravity while swinging, all contexts (1.25x).
MotionParams*.GravityParams.GravityFall* *= 1.25

[configs\hero\hero_webzipsetup_ground.config]
